    def validate_vision_record(value: Mapping[str, Any]) -> VisionRecord:
        # require every approved field; reject invalid enum, UUID, SHA-256,
        # RFC3339 timestamp, confidence, unusable-image, and first-image values
        missing = REQUIRED_FIELDS - value.keys()
        if missing or not _valid_types_and_enums(value):
            raise VisionValidationError("INVALID_VISION_RECORD")
        if value["image_quality"] == "unusable" and not _valid_unusable(value):
            raise VisionValidationError("INVALID_VISION_RECORD")
        if value["previous_image_id"] is None and value["change_vs_previous"] != "unknown":
            raise VisionValidationError("INVALID_VISION_RECORD")
        return VisionRecord.from_mapping(value)

    class OpenCvRtspFrameCapture:
        def capture_one(self) -> CapturedFrame:
            # open VideoCapture; read one frame; JPEG encode; release in finally
            # map open, read, and encode errors to sanitized CaptureError codes
            capture = cv2.VideoCapture(self._rtsp_url)
            try:
                if not capture.isOpened():
                    raise CaptureError("RTSP_OPEN_FAILED")
                ok, frame = capture.read()
                if not ok or frame is None:
                    raise CaptureError("RTSP_READ_FAILED")
                ok, encoded = cv2.imencode(".jpg", frame)
                if not ok:
                    raise CaptureError("JPEG_ENCODE_FAILED")
                return CapturedFrame(jpeg_bytes=encoded.tobytes(), captured_at=utc_now())
            finally:
                capture.release()

    class ImageStore:
        def save(self, jpeg_bytes: bytes, captured_at: datetime) -> ImageEvidence:
            # reject an empty or undecodable JPEG, atomically write date/image_id.jpg,
            # calculate SHA-256, and return only a relative path
            decoded = cv2.imdecode(np.frombuffer(jpeg_bytes, np.uint8), 1)
            if not jpeg_bytes or decoded is None:
                raise CaptureError("JPEG_INVALID")
            image_id = str(uuid.uuid4())
            relative = Path("images") / captured_at.date().isoformat() / f"{image_id}.jpg"
            target = self._root / relative
            target.parent.mkdir(parents=True, exist_ok=True)
            temporary = target.with_suffix(".tmp")
            temporary.write_bytes(jpeg_bytes)
            os.replace(temporary, target)
            return ImageEvidence(image_id, relative.as_posix(), sha256(jpeg_bytes).hexdigest())

    def analyze(self, evidence: ImageEvidence, previous_image_id: str | None) -> Mapping[str, Any]:
        # call one OpenAI-compatible chat completion through injected requests.Session
        # parse JSON text only; raise MODEL_TIMEOUT, MODEL_REQUEST_FAILED,
        # MODEL_INVALID_JSON, or MODEL_INVALID_RESPONSE; never fabricate a result
        try:
            response = self._session.post(self._url, headers=self._headers,
                                          json=self._payload(evidence, previous_image_id),
                                          timeout=self._timeout_seconds)
            response.raise_for_status()
        except requests.Timeout as error:
            raise AnalysisError("MODEL_TIMEOUT") from error
        except requests.RequestException as error:
            raise AnalysisError("MODEL_REQUEST_FAILED") from error
        try:
            return json.loads(response.json()["choices"][0]["message"]["content"])
        except (KeyError, IndexError, TypeError, ValueError) as error:
            raise AnalysisError("MODEL_INVALID_JSON") from error

    def capture_and_analyze_once() -> CaptureOutcome:
        # load environment settings; capture a frame; save it; locate the prior
        # same-device record; analyze; validate; atomically persist only valid records
        try:
            frame = service.capture.capture_one()
        except CaptureError as error:
            return service.persist_capture_failure(error.code)
        evidence = service.images.save(frame.jpeg_bytes, frame.captured_at)
        try:
            raw = service.analyzer.analyze(evidence, service.previous_image_id())
            record = validate_vision_record(service.attach_evidence(raw, evidence, frame.captured_at))
        except (AnalysisError, VisionValidationError) as error:
            return service.persist_analysis_failure(evidence, error.code)
        service.persist_record(record)
        status = "image_unusable" if record.image_quality == "unusable" else "success"
        return CaptureOutcome(status, evidence.image_id, record.to_mapping(), None)

## Revision 1: two plant zones and fuller observations

Requested in one pass after the first real smoke test, so this stays a single Issue #12 change rather than follow-up tasks.

- Why: the frame contains two plants whose foliage overlaps, so judging the picture as one object is wrong; and the first field set could not express common observations (browning, curl, spots, leaf loss, stem posture, occlusion) or admit that the model saw nothing.
- Zones: `config/vision_zones.example.json` plus `SOIL3_VISION_ZONES_PATH`. One normalized `[x, y, w, h]` rect per zone. The whole frame is still stored under `frames/`; only the model input is cropped. History, records, and failures are keyed by `plant_zone`, and a zone is compared only with its own earlier crop.
- Observations: 17 returned fields, all evidence-limited fields nullable, and `overall_visual_state` reduced to appearance (`normal / mild_abnormality / obvious_abnormality / severe_abnormality / unavailable`). Missing evidence must be `null`, never `false` or `none`; the validator rejects a state that cites no observation.
- Failure isolation: one zone failing no longer discards the other zone's record; the run reports `partial`.
- Out of scope, recorded as known follow-up rather than silently fixed here: failure records still do not carry the provider HTTP status or error code, which made the first quota block slower to diagnose than it should have been.
